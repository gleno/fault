package fault

import (
	"strings"
	"testing"
)

func TestGetFullErrorNoMessageDuplication(t *testing.T) {
	chain := From(From(Errorf("DEEPESTTOKEN"), "midmsg"), "topmsg")

	full := GetFullError(chain)

	deepCount := strings.Count(full, "DEEPESTTOKEN")
	midCount := strings.Count(full, "midmsg")

	t.Logf("GetFullError output:\n%s", full)
	t.Logf("DEEPESTTOKEN appears %d times, midmsg appears %d times", deepCount, midCount)

	if deepCount != 1 {
		t.Errorf("deepest token duplicated: appears %d times, want 1", deepCount)
	}
	if midCount != 1 {
		t.Errorf("mid message duplicated: appears %d times, want 1", midCount)
	}
}

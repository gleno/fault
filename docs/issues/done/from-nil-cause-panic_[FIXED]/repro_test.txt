package fault

import (
	"testing"
)

// TestPackageFromNilCause: From(nil, msg) stores a nil cause; Error() then
// dereferences it (error_cause.go:13) and panics.
func TestPackageFromNilCause(t *testing.T) {
	f := From(nil, "wrapping context")
	_ = f.Error() // panics: nil pointer dereference
}

// TestMethodFromNilCause: same defect class via a builder method. Wrap(nil, _)
// returns nil, which is stored as the cause, then dereferenced in Error().
func TestMethodFromNilCause(t *testing.T) {
	s := Sentinel("base")
	f := s.From(nil, "wrapping context")
	_ = f.Error() // panics: nil pointer dereference
}

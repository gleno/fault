package fault

import (
	"errors"
	"testing"
)

// Two DISTINCT _ErrorWithCause with identical msg => Is returns true (message-equality).
// The analogous pair of DISTINCT _ErrorWithTag with identical msg SHOULD behave the same
// for consistency, since _ErrorWithTag embeds _ErrorWithCause and inherits its Is.
func TestTagIsConsistencyWithCause(t *testing.T) {
	// Cause-only pair, identical messages.
	causeA := From(Sentinel("inner"), "same msg")
	causeB := From(Sentinel("inner"), "same msg")
	causeMatch := errors.Is(causeA, causeB)
	t.Logf("errors.Is(causeA, causeB) [both _ErrorWithCause]=%v", causeMatch)

	// Tagged pair, identical messages, identical tag.
	taggedA := TagAsUserFault(Sentinel("inner"), "same msg")
	taggedB := TagAsUserFault(Sentinel("inner"), "same msg")
	taggedMatch := errors.Is(taggedA, taggedB)
	t.Logf("errors.Is(taggedA, taggedB) [both _ErrorWithTag]=%v", taggedMatch)

	// The _ErrorWithCause.Is does target.(*_ErrorWithCause); a *_ErrorWithTag target
	// fails that assertion. So taggedA.Is(taggedB) -> false even though messages match.
	if causeMatch != taggedMatch {
		t.Errorf("INCONSISTENCY: cause-pair Is=%v but tagged-pair Is=%v (same-msg semantics differ by type)", causeMatch, taggedMatch)
	}
}
